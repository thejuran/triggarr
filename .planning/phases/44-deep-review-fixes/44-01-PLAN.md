    ```python
    def _parse_version(version_str: str) -> tuple[int, ...]:
        try:
            cleaned = version_str.lstrip("v")
            if not cleaned:
                return (0,)
            parts = []
            for part in cleaned.split("."):
                m = re.match(r"^(\d+)", part)
                if m:
                    parts.append(int(m.group(1)))
            return tuple(parts) if parts else (0,)
        except (ValueError, AttributeError):
            return (0,)
    ```
