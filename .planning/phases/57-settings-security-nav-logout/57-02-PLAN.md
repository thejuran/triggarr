    <script>
    function updateMethodWarning(value) {
        var el = document.getElementById('method-warning');
        if (value === 'External') {
            el.textContent = 'Login page will be bypassed. Ensure your reverse proxy handles auth.';
            el.style.display = 'block';
        } else if (value === 'Basic') {
            el.textContent = 'Browser will show a native popup instead of the login page.';
            el.style.display = 'block';
        } else {
            el.style.display = 'none';
        }
    }
    // Initialize warning on page load if a non-Forms method is already selected
    (function() {
        var select = document.querySelector('select[name="auth_method"]');
        if (select) updateMethodWarning(select.value);
    })();

    function toggleApiKeyVisibility() {
        var input = document.getElementById('api-key-input');
        var showIcon = document.getElementById('eye-icon-show');
        var hideIcon = document.getElementById('eye-icon-hide');
        if (input.type === 'password') {
            input.type = 'text';
            showIcon.classList.add('hidden');
            hideIcon.classList.remove('hidden');
        } else {
            input.type = 'password';
            showIcon.classList.remove('hidden');
            hideIcon.classList.add('hidden');
        }
    }

    function copyApiKey(btn) {
        var key = document.getElementById('api-key-input').dataset.key;
        if (navigator.clipboard && navigator.clipboard.writeText) {
            navigator.clipboard.writeText(key).then(function() {
                btn.innerHTML = '<span class="text-sm">Copied!</span>';
                setTimeout(function() {
                    btn.innerHTML = '<svg xmlns="http://www.w3.org/2000/svg" width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="currentColor" stroke-width="2" stroke-linecap="round" stroke-linejoin="round"><rect x="9" y="9" width="13" height="13" rx="2" ry="2"/><path d="M5 15H4a2 2 0 0 1-2-2V4a2 2 0 0 1 2-2h9a2 2 0 0 1 2 2v1"/></svg>';
                }, 2000);
            });
        }
    }

    function showRegenConfirm() {
        document.getElementById('regen-confirm').classList.remove('hidden');
        document.getElementById('regen-btn').classList.add('hidden');
    }

    function cancelRegen() {
        document.getElementById('regen-confirm').classList.add('hidden');
        document.getElementById('regen-btn').classList.remove('hidden');
    }

    function confirmRegen() {
        htmx.ajax('POST', '{{ request.url_for("regenerate_api_key_endpoint") }}', {
            target: '#apikey-section',
            swap: 'innerHTML'
        });
    }
    </script>
