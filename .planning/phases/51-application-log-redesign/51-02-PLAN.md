```python
@router.get("/partials/log-viewer", response_class=HTMLResponse)
async def partial_log_viewer(request: Request, level: str = "") -> HTMLResponse:
    """Return an HTML fragment for the application log viewer (htmx partial)."""
    _VALID_LEVELS = {"ERROR", "WARNING", "INFO", "DEBUG"}
    log_entries = log_buffer.get_recent(30)
    # Server-side level filter (per D-06, T-51-01)
    if level.upper() in _VALID_LEVELS:
        level = level.upper()
        log_entries = [e for e in log_entries if e.level == level]
    else:
        level = ""
    return templates.TemplateResponse(
        request=request,
        name="partials/log_viewer.html",
        context={"log_entries": log_entries, "selected_level": level},
    )
```

Key points:
- `_VALID_LEVELS` whitelist prevents arbitrary input (per T-51-01 threat mitigation)
- Invalid level values silently fall through to "All" (empty string)
- `selected_level` passed to template so the dropdown renders with correct selection after swap
- `level.upper()` normalizes case

Add the following script block:

```html
<script>
/* Log viewer controls — lives outside partial to survive htmx outerHTML swaps */

/* D-04: Auto-scroll log body to bottom after every htmx swap */
document.addEventListener('htmx:afterSwap', function(evt) {
  if (evt.detail.target && evt.detail.target.id === 'log-viewer') {
    var logBody = evt.detail.target.querySelector('.log-body');
    if (logBody) logBody.scrollTop = logBody.scrollHeight;
    /* Re-apply expanded state from body if needed (per D-02, LOG-05/LOG-06) */
    if (document.body.classList.contains('log-expanded')) {
      evt.detail.target.classList.add('expanded');
    }
    /* Re-apply paused state from body (per D-05) */
    if (document.body.hasAttribute('data-log-paused')) {
      var viewer = evt.detail.target;
      viewer.setAttribute('hx-trigger', 'none');
      htmx.process(viewer);
      var pauseBtn = viewer.querySelector('[data-pause-btn]');
      if (pauseBtn) pauseBtn.classList.add('text-yellow-400');
    }
  }
});

/* D-02, D-03: Toggle expand/collapse terminal pane */
function toggleLogExpand() {
  var viewer = document.getElementById('log-viewer');
  if (!viewer) return;
  viewer.classList.toggle('expanded');
  document.body.classList.toggle('log-expanded');
  var logBody = viewer.querySelector('.log-body');
  if (logBody) logBody.scrollTop = logBody.scrollHeight;
}

/* D-05: Toggle pause/resume htmx polling */
function toggleLogPause(btn) {
  var viewer = document.getElementById('log-viewer');
  if (!viewer) return;
  var isPaused = document.body.hasAttribute('data-log-paused');
  if (isPaused) {
    document.body.removeAttribute('data-log-paused');
    viewer.setAttribute('hx-trigger', 'every 5s');
    htmx.process(viewer);
    btn.classList.remove('text-yellow-400');
    btn.title = 'Pause stream';
  } else {
    document.body.setAttribute('data-log-paused', '');
    viewer.setAttribute('hx-trigger', 'none');
    htmx.process(viewer);
    btn.classList.add('text-yellow-400');
    btn.title = 'Resume stream';
  }
}

/* Auto-scroll on initial page load */
document.addEventListener('DOMContentLoaded', function() {
  var logBody = document.querySelector('#log-viewer .log-body');
  if (logBody) logBody.scrollTop = logBody.scrollHeight;
});
</script>
```
