    ```python
    def _safe_next_url(next_param: str | None) -> str:
        """Validate ?next= param, rejecting open redirects. Returns safe URL or '/'."""
        if not next_param:
            return "/"
        if next_param.startswith(("http://", "https://", "//")):
            return "/"
        if "\\" in next_param:
            return "/"
        if not next_param.startswith("/"):
            return "/"
        return next_param
    ```
