**Extend `deduplicate_to_seasons`** to count episodes per season:
```python
def deduplicate_to_seasons(episodes: list[dict]) -> list[dict]:
    seen: dict[tuple[int, int], dict] = {}  # Change from set to dict to accumulate counts
    seasons: list[dict] = []
    for ep in episodes:
        series_id = ep.get("seriesId")
        season_number = ep.get("seasonNumber")
        if series_id is None or season_number is None:
            continue
        key = (series_id, season_number)
        if key not in seen:
            title = ep.get("series", {}).get("title", f"Series {series_id}")
            entry = {
                "seriesId": series_id,
                "seasonNumber": season_number,
                "display_name": f"{title} - Season {season_number}",
                "episode_count": 1,
            }
            seen[key] = entry
            seasons.append(entry)
        else:
            seen[key]["episode_count"] += 1
    return seasons
```

Then in Sonarr insert_search_entry calls, pass `missing_count=season["episode_count"]`:
